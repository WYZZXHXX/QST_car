\\OLED板同步打印
#include <stdio.h>
#include <unistd.h>
#include "ohos_init.h"
#include "cmsis_os2.h"
#include "hal_bsp_ap3216c.h"
#include "hal_bsp_ssd1306.h"

#define SENSOR_TASK_STACK_SIZE 2048
#define OLED_TEXT_SIZE 16

static void SensorTask(void);

static void DisplaySensorData(uint16_t ir, uint16_t als, uint16_t ps)
{
    uint8_t line[22];

    SSD1306_CLS();

    SSD1306_ShowStr(8, 0, (uint8_t *)"AP3216C SENSOR", OLED_TEXT_SIZE);

    (void)snprintf((char *)line, sizeof(line), "IR :%u", (unsigned int)ir);
    SSD1306_ShowStr(16, 1, line, OLED_TEXT_SIZE);

    (void)snprintf((char *)line, sizeof(line), "ALS:%u", (unsigned int)als);
    SSD1306_ShowStr(16, 2, line, OLED_TEXT_SIZE);

    (void)snprintf((char *)line, sizeof(line), "PS :%u", (unsigned int)ps);
    SSD1306_ShowStr(16, 3, line, OLED_TEXT_SIZE);
}

static void I2cAp3216cDemo(void)
{
    osThreadAttr_t options = {
        .name = "sensor_task",
        .attr_bits = 0,
        .cb_mem = NULL,
        .cb_size = 0,
        .stack_mem = NULL,
        .stack_size = SENSOR_TASK_STACK_SIZE,
        .priority = osPriorityNormal,
    };

    osThreadId_t taskId = osThreadNew((osThreadFunc_t)SensorTask, NULL, &options);
    if (taskId == NULL) {
        printf("Failed to create sensor task!\r\n");
    } else {
        printf("Create sensor task OK!\r\n");
    }
}

static void SensorTask(void)
{
    uint16_t ir = 0;
    uint16_t als = 0;
    uint16_t ps = 0;
    uint32_t result;

    result = AP3216C_Init();
    if (result != 0) {
        printf("AP3216C init failed: 0x%x\r\n", result);
        return;
    }
    printf("AP3216C initialized.\r\n");

    usleep(10000);

    result = SSD1306_Init();
    if (result != 0) {
        printf("SSD1306 init failed: 0x%x\r\n", result);
        return;
    }
    printf("SSD1306 initialized.\r\n");

    SSD1306_CLS();
    SSD1306_ShowStr(0, 1, (uint8_t *)"  STARTING...", OLED_TEXT_SIZE);
    sleep(1);

    while (1) {
        result = AP3216C_ReadData(&ir, &als, &ps);
        if (result == 0) {
            printf("IR=%u  ALS=%u  PS=%u\r\n",
                (unsigned int)ir, (unsigned int)als, (unsigned int)ps);
            DisplaySensorData(ir, als, ps);
        } else {
            printf("AP3216C read failed: 0x%x\r\n", result);
        }
        sleep(1);
    }
}

APP_FEATURE_INIT(I2cAp3216cDemo);






\\加上光照传感功能
#include <stdio.h>
#include <unistd.h>
#include "ohos_init.h"
#include "cmsis_os2.h"
#include "hal_bsp_ap3216c.h"
#include "ssd1306_oled.h"
#include "wifiiot_gpio.h"
#include "wifiiot_gpio_ex.h"

#define SENSOR_TASK_STACK_SIZE 4096
#define OLED_TEXT_SIZE 16

#define LED_GPIO_IDX        WIFI_IOT_GPIO_IDX_6
#define LED_GPIO_FUNC       WIFI_IOT_IO_FUNC_GPIO_6_GPIO
#define ALS_DARK_THRESHOLD  100

static void SensorTask(void);
static uint8_t ledOn = 0;

static void LedInit(void)
{
    IoSetFunc(WIFI_IOT_IO_NAME_GPIO_6, LED_GPIO_FUNC);
    GpioSetDir(LED_GPIO_IDX, WIFI_IOT_GPIO_DIR_OUT);
    GpioSetOutputVal(LED_GPIO_IDX, WIFI_IOT_GPIO_VALUE0);
}

static void DisplaySensorData(uint16_t ir, uint16_t als, uint16_t ps)
{
    uint8_t line[22];

    SSD1306_CLS();

    if (ledOn) {
        SSD1306_ShowStr(4, 0, (uint8_t *)"AP3216C   LED:ON", OLED_TEXT_SIZE);
    } else {
        SSD1306_ShowStr(4, 0, (uint8_t *)"AP3216C  LED:OFF", OLED_TEXT_SIZE);
    }

    (void)snprintf((char *)line, sizeof(line), "IR :%u", (unsigned int)ir);
    SSD1306_ShowStr(16, 1, line, OLED_TEXT_SIZE);

    (void)snprintf((char *)line, sizeof(line), "ALS:%u", (unsigned int)als);
    SSD1306_ShowStr(16, 2, line, OLED_TEXT_SIZE);

    (void)snprintf((char *)line, sizeof(line), "PS :%u", (unsigned int)ps);
    SSD1306_ShowStr(16, 3, line, OLED_TEXT_SIZE);
}

static void I2cAp3216cDemo(void)
{
    osThreadAttr_t options = {
        .name = "sensor_task",
        .attr_bits = 0,
        .cb_mem = NULL,
        .cb_size = 0,
        .stack_mem = NULL,
        .stack_size = SENSOR_TASK_STACK_SIZE,
        .priority = osPriorityNormal,
    };

    osThreadId_t taskId = osThreadNew((osThreadFunc_t)SensorTask, NULL, &options);
    if (taskId == NULL) {
        printf("Failed to create sensor task!\r\n");
    } else {
        printf("Create sensor task OK!\r\n");
    }
}

static void SensorTask(void)
{
    uint16_t ir = 0;
    uint16_t als = 0;
    uint16_t ps = 0;
    uint32_t result;

    result = AP3216C_Init();
    if (result != 0) {
        printf("AP3216C init failed: 0x%x\r\n", result);
        return;
    }
    printf("AP3216C initialized.\r\n");

    usleep(10000);

    result = SSD1306_Init();
    if (result != 0) {
        printf("SSD1306 init failed: 0x%x\r\n", result);
        return;
    }
    printf("SSD1306 initialized.\r\n");

    LedInit();
    printf("LED GPIO_6 OK.\r\n");

    SSD1306_CLS();
    SSD1306_ShowStr(0, 1, (uint8_t *)"  STARTING...", OLED_TEXT_SIZE);
    sleep(1);

    while (1) {
        result = AP3216C_ReadData(&ir, &als, &ps);
        if (result == 0) {
            if (als < ALS_DARK_THRESHOLD) {
                if (!ledOn) {
                    GpioSetOutputVal(LED_GPIO_IDX, WIFI_IOT_GPIO_VALUE1);
                    ledOn = 1;
                }
            } else {
                if (ledOn) {
                    GpioSetOutputVal(LED_GPIO_IDX, WIFI_IOT_GPIO_VALUE0);
                    ledOn = 0;
                }
            }

            printf("IR=%u ALS=%u PS=%u LED=%s\r\n",
                (unsigned int)ir, (unsigned int)als, (unsigned int)ps,
                ledOn ? "ON" : "OFF");
            DisplaySensorData(ir, als, ps);
        } else {
            printf("AP3216C read failed: 0x%x\r\n", result);
        }
        sleep(1);
    }
}

APP_FEATURE_INIT(I2cAp3216cDemo);
