#include "stm32f10x.h"
#include "sys.h"
#include "motor.h"

int main(void)
{
    u8 i;

    Stm32_Clock_Init(9);                // 系统时钟 72MHz
    MY_NVIC_PriorityGroupConfig(2);     // 中断优先级分组
    uart_init(115200);                  // 串口 115200
    JTAG_Set(JTAG_SWD_DISABLE);         // 关闭 JTAG
    JTAG_Set(SWD_ENABLE);               // 打开 SWD

    PWM_Init(7199, 9);                  // 电机 PWM 初始化
    colorful_led_Init();                // 炫彩灯初始化

    // ★ 左灯:先设颜色,再刷新
    for (i = 1; i <= led_num; i++)
        L_ws2812_rgb(i, WS_BLUE);       // 左灯蓝色
    L_ws2812_refresh(led_num);

    // ★ 右灯:先设颜色,再刷新
    for (i = 1; i <= led_num; i++)
        R_ws2812_rgb(i, WS_BLUE);        // 右灯红色
    R_ws2812_refresh(led_num);

    while (1)
    {
        Set_Pwm(2500, 2500);            // 电机前进
        delay_ms(5000);
	Set_Pwm(-2500, -2500);            // 电机后退
        delay_ms(5000);
    }
}
